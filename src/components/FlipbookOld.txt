"use client";
import React, { useRef } from 'react';
import HTMLFlipBook from 'react-pageflip';

export default function Flipbook({ pages, isLandscape }: { pages: string[], isLandscape: boolean }) {
  const flipBookRef = useRef<any>(null);

  // ✅ ปรับอัตราส่วนให้เหมาะสม (สัดส่วนมาตรฐานกระดาษ A4/Letter)
  // แนวตั้ง 3:4.2 (ขยายความสูงเพิ่ม) | แวนอน 4.2:3 (ขยายความกว้างเพิ่ม)
  const width = isLandscape ? 400 : 300;
  const height = isLandscape ? 300 : 400; 

  const playFlipSound = () => {
    const audio = new Audio('https://assets.mixkit.co');
    audio.volume = 0.15;
    audio.play().catch(() => {});
  };

  if (pages.length === 0) return null;

  return (
    // ✅ ลด padding (p-1) เพื่อให้หนังสือเข้าใกล้ขอบจอมากขึ้น
    <div className="w-full h-full flex flex-col items-center justify-center overflow-hidden p-1 sm:p-2">
      
      <div className="relative w-full h-full flex items-center justify-center">
        {/* @ts-ignore */}
        <HTMLFlipBook
          width={width}
          height={height}
          size="stretch"         // ✅ กลับมาใช้ stretch แต่คุมด้วย Container h-full
          minWidth={200}
          maxWidth={2000}
          minHeight={200}
          maxHeight={2000}
          
          usePortrait={!isLandscape}
          startPage={0}
          drawShadow={true}
          flippingTime={800}     // ปรับความเร็วการพลิกให้ดูพรีเมียมขึ้น
          showCover={true}
          mobileScrollSupport={true}
          onFlip={playFlipSound}
          className="shadow-[0_20px_50px_rgba(0,0,0,0.3)] mx-auto" // เพิ่มเงาให้ดูมีมิติ
          ref={flipBookRef}
        >
          {pages.map((image, index) => (
            <div key={index} className="bg-white flex items-center justify-center w-full h-full border-l border-gray-100">
              <img 
                src={image} 
                alt={`Page ${index + 1}`} 
                // ✅ ใช้ object-fill ถ้าต้องการให้เต็มกระดาษหนังสือ หรือ object-contain เพื่อความเป๊ะ
                className="w-full h-full object-contain pointer-events-none" 
              />
            </div>
          ))}
        </HTMLFlipBook>
      </div>

      {/* แถบสถานะแบบจางๆ ไม่รบกวนสายตา */}
      <div className="absolute bottom-4 left-1/2 -translate-x-1/2 px-3 py-1 bg-black/5 rounded-full text-[10px] text-gray-400 font-medium backdrop-blur-sm">
        {pages.length} Pages • Click corner to flip
      </div>
    </div>
  );
}

"use client";
import React, { useRef } from 'react';
import HTMLFlipBook from 'react-pageflip';

export default function Flipbook({ pages, isLandscape }: { pages: string[], isLandscape: boolean }) {
  const flipBookRef = useRef<any>(null);

  // ✅ แก้ไข: ใช้สัดส่วนกระดาษที่แน่นอน (Ratio)

  const width = isLandscape ? 400 : 300;
  const height = isLandscape ? 300 : 400;

  const playFlipSound = () => {
    const audio = new Audio('https://assets.mixkit.co');
    audio.volume = 0.2;
    audio.play().catch(() => {});
  };

  if (pages.length === 0) return null;

  return (
    // ✅ ใช้ flex-col และ h-full เพื่อให้ Container คุมพื้นที่ทั้งหมด
    <div className="w-full h-full flex flex-col items-center justify-center overflow-hidden p-2">
      
      <div className="relative w-full h-full flex items-center justify-center">
        {/* @ts-ignore */}
        <HTMLFlipBook
          width={width}
          height={height}
          size="fixed" // ✅ เปลี่ยนจาก stretch เป็น fixed เพื่อให้รักษาสัดส่วนไม่ให้หลุดขอบ
          minWidth={200}
          maxWidth={2000}
          minHeight={200}
          maxHeight={2000}
          
          // ✅ ตั้งค่าให้พอดีหน้าจออัตโนมัติ
          usePortrait={!isLandscape}
          startPage={0}
          drawShadow={true}
          showCover={true}
          mobileScrollSupport={true}
          onFlip={playFlipSound}
          className="shadow-2xl mx-auto"
          ref={flipBookRef}
          style={{ margin: 'auto' }} // จัดกึ่งกลางเป๊ะๆ
        >
          {pages.map((image, index) => (
            <div key={index} className="bg-white flex items-center justify-center overflow-hidden border border-gray-100">
              <img 
                src={image} 
                alt={`Page ${index + 1}`} 
                // ✅ ใช้ h-full w-auto และ object-contain เพื่อให้รูปไม่ล้นขอบบนล่าง
                className="max-w-full max-h-full w-auto h-auto object-contain pointer-events-none" 
              />
            </div>
          ))}
        </HTMLFlipBook>
      </div>

      {/* ✅ เพิ่มแถบช่วยบอกเลขหน้าด้านล่างเพื่อให้รู้ว่าไม่หลุดจอ */}
      <div className="mt-2 text-[10px] text-gray-500 font-bold uppercase tracking-widest">
        Click corner to flip • {isLandscape ? 'Landscape Mode' : 'Portrait Mode'}
      </div>
    </div>
  );
}

"use client";
import React, { useRef, useEffect } from 'react';
import HTMLFlipBook from 'react-pageflip';

export default function Flipbook({ pages, isLandscape }: { pages: string[], isLandscape: boolean }) {
  const flipBookRef = useRef<any>(null);

  // กำหนดอัตราส่วนหน้ากระดาษ (Ratio)
  // แนวตั้ง: กว้าง 700 สูง 1000 | แนวนอน: กว้าง 1000 สูง 700
  const width = isLandscape ? 1000 : 700;
  const height = isLandscape ? 700 : 1000;

  const playFlipSound = () => {
    const audio = new Audio('https://assets.mixkit.co');
    audio.volume = 0.2;
    audio.play().catch(() => {});
  };

  if (pages.length === 0) {
    return <div className="p-20 text-gray-400 font-bold">กรุณาอัปโหลดไฟล์ PDF เพื่อเริ่มใช้งาน</div>;
  }

  return (
    <div className="w-full h-[85vh] flex items-center justify-center p-2 md:p-6 bg-slate-200 overflow-hidden">
      <div className="w-full h-full max-w-7xl flex items-center justify-center">
        {/* @ts-ignore */}
        <HTMLFlipBook
          width={width}
          height={height}
          size="stretch"         // บังคับให้ขยายเต็มพื้นที่ Container
          minWidth={200}
          maxWidth={2000}
          minHeight={200}
          maxHeight={2000}
          drawShadow={true}
          maxShadowOpacity={0.4}
          showCover={true}
          mobileScrollSupport={true}
          onFlip={playFlipSound}
          className="shadow-2xl bg-white"
          ref={flipBookRef}
          // บังคับให้เริ่มหน้า 1 เสมอเมื่อเปลี่ยนโหมด
          startPage={0}
        >
          {pages.map((image, index) => (
            <div key={index} className="bg-white flex items-center justify-center w-full h-full border-l border-gray-100">
              <img 
                src={image} 
                alt={`Page ${index + 1}`} 
                className="w-full h-full object-contain pointer-events-none" 
                />
            </div>
          ))}
        </HTMLFlipBook>
      </div>
    </div>
  );
}
